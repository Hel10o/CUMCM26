def H(data): return hashlib.sha256(data).hexdigest()
def FH(path):

def numbers(s):

def prose(s):

def sentences(s):

 return [norm(t) for p in re.split(r'\r?\n\s*\r?\n',prose(s)) for t in re.findall(r'[^。！？]+[。！？]',p) if norm(t)]
